public class BanListaRepozitorijumEF : IBanListaRepository
{
    private readonly string _konekcija;

    public BanListaRepozitorijumEF(string konekcija)
    {
        _konekcija = konekcija;
    }

    public List<BanListaKlasa> DajSvuBanListu()
    {
        try
        {
            using (var dbContext = new YuGiOhDBEntities1())
            {
                // LINQ upit umesto SQL
                var bannedCards = dbContext.BannedKarte
                    .OrderByDescending(k => k.DatumBana)
                    .ToList();

                // Mapiranje Entity Framework klasa u domenskih klasa
                return bannedCards.Select(k => new BanListaKlasa
                {
                    BanListaID = k.BannedKartaID,
                    NazivKarte = k.NazivKarte,
                    DatumDodavanja = k.DatumBana ?? DateTime.Now
                }).ToList();
            }
        }
        catch
        {
            return new List<BanListaKlasa>();
        }
    }

    public List<BanListaKlasa> DajBanListuSudije(int sudijaID)
    {
        try
        {
            using (var dbContext = new YuGiOhDBEntities1())
            {
                // LINQ WHERE filter
                var bannedCards = dbContext.BannedKarte
                    .Where(k => k.SudijaID == sudijaID)
                    .OrderByDescending(k => k.DatumBana)
                    .ToList();

                return bannedCards.Select(k => new BanListaKlasa
                {
                    BanListaID = k.BannedKartaID,
                    SudijaID = k.SudijaID ?? 0,
                    NazivKarte = k.NazivKarte,
                    DatumDodavanja = k.DatumBana ?? DateTime.Now
                }).ToList();
            }
        }
        catch
        {
            return new List<BanListaKlasa>();
        }
    }

    public int DodajNaBanListu(int sudijaID, string nazivKarte)
    {
        try
        {
            using (var dbContext = new YuGiOhDBEntities1())
            {
                var novaBannedKarta = new BannedKarta
                {
                    SudijaID = sudijaID,
                    NazivKarte = nazivKarte,
                    Razlog = "Dodala sudija",
                    DatumBana = DateTime.Now
                };

                dbContext.BannedKarte.Add(novaBannedKarta);
                dbContext.SaveChanges();

                return novaBannedKarta.BannedKartaID;
            }
        }
        catch
        {
            return 0;
        }
    }

    public int ObrisiSaBanListe(int banListaID)
    {
        try
        {
            using (var dbContext = new YuGiOhDBEntities1())
            {
                var kartaZaBrisanje = dbContext.BannedKarte
                    .FirstOrDefault(k => k.BannedKartaID == banListaID);

                if (kartaZaBrisanje != null)
                {
                    dbContext.BannedKarte.Remove(kartaZaBrisanje);
                    dbContext.SaveChanges();
                    return 1;
                }

                return 0;
            }
        }
        catch
        {
            return 0;
        }
    }

    public bool DaLiJeKartaNaBanListi(int sudijaID, string nazivKarte)
    {
        try
        {
            using (var dbContext = new YuGiOhDBEntities1())
            {
                // LINQ Any() umesto SQL COUNT
                bool postoji = dbContext.BannedKarte
                    .Any(k => k.SudijaID == sudijaID && k.NazivKarte == nazivKarte);

                return postoji;
            }
        }
        catch
        {
            return false;
        }
    }
}
